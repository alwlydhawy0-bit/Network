! ---------------------------------------------------------------------
! Inter-VLAN security ACLs - applied INBOUND on each client SVI.
! Full rule-by-rule rationale: docs/05-security-acl-policy.md
! Server objects: .10 SRV-DC01 (DNS/DHCP/AAA)  .20 SRV-WEB
!                 .30 SRV-FILE (FTP)           .40 SRV-FINAPP (HTTPS)
! ---------------------------------------------------------------------
ip access-list extended VLAN10_MGMT_IN
 remark MANAGEMENT - administrators have full access; anti-spoofing only
 permit ip 192.168.10.0 0.0.0.255 any
 deny ip any any
!
ip access-list extended VLAN20_IT_IN
 remark DHCP discover/request (relayed to SRV-DC01)
 permit udp any eq bootpc any eq bootps
 remark Replies to sessions started by MANAGEMENT
 permit icmp 192.168.20.0 0.0.0.255 192.168.10.0 0.0.0.255 echo-reply
 permit tcp 192.168.20.0 0.0.0.255 192.168.10.0 0.0.0.255 established
 remark No other access to the MANAGEMENT VLAN or to device administration
 deny ip any 192.168.10.0 0.0.0.255
 deny tcp any 192.168.0.0 0.0.255.255 eq 22
 deny tcp any 192.168.0.0 0.0.255.255 eq telnet
 deny ip any 10.0.0.0 0.255.255.255
 remark Finance application is restricted to FINANCE and EXECUTIVE
 deny tcp any host 192.168.60.40
 remark IT support may reach all other internal VLANs, servers and the Internet
 permit ip 192.168.20.0 0.0.0.255 any
 deny ip any any
!
ip access-list extended VLAN30_HR_IN
 remark DHCP discover/request (relayed to SRV-DC01)
 permit udp any eq bootpc any eq bootps
 remark Ping own default gateway
 permit icmp 192.168.30.0 0.0.0.255 host 192.168.30.1 echo
 remark Replies to MANAGEMENT and IT support sessions
 permit icmp 192.168.30.0 0.0.0.255 192.168.10.0 0.0.0.255 echo-reply
 permit icmp 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255 echo-reply
 permit tcp 192.168.30.0 0.0.0.255 192.168.10.0 0.0.0.255 established
 permit tcp 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255 established
 remark Approved services - DNS, intranet web, file server, ping servers
 permit udp 192.168.30.0 0.0.0.255 host 192.168.60.10 eq domain
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.60.10 eq domain
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.60.20 eq www
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.60.20 eq 443
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.60.30 eq ftp
 permit icmp 192.168.30.0 0.0.0.255 192.168.60.0 0.0.0.255 echo
 remark Block every other internal and private destination
 deny ip any 192.168.0.0 0.0.255.255
 deny ip any 10.0.0.0 0.255.255.255
 deny ip any 172.16.0.0 0.15.255.255
 remark Internet access (all private ranges are already denied above)
 permit ip 192.168.30.0 0.0.0.255 any
 deny ip any any
!
